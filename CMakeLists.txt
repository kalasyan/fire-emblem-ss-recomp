# fire-emblem-ss-recomp — native build of Fire Emblem: The Sacred Stones
# (fireemblem8u decomp) for PC/Android.
#
#   cmake -B build -G Ninja -DFE8_HOST=headless   # or sdl
#   cmake --build build
#
# The GBA ROM build is still the upstream Makefile; this file only drives the
# native port. No ROM is needed to build: game data is read in place from the
# player's own ROM, which the port maps at its GBA address (see docs/PORT.md).
cmake_minimum_required(VERSION 3.22)
project(fire_emblem_ss_recomp C ASM)

set(FE8_REGION "us" CACHE STRING "Game region: us, eu")
set(FE8_HOST "sdl" CACHE STRING "Frontend: sdl or headless")
# The decomp is written for agbcc and is very noisy under modern compilers.
set(FE8_GAME_WARNINGS "-w" CACHE STRING "Warning flags for the game code")
# Where the game image is loaded. It must sit below 4 GiB, next to the GBA
# memory map (0x02000000-0x0E010000): the game stores pointers in 32-bit
# fields, as on the GBA.
set(FE8_IMAGE_BASE "0x10000000" CACHE STRING "Load address of the game image")

find_package(Python3 REQUIRED COMPONENTS Interpreter)

set(FE8_SDL3_TAG "release-3.4.18" CACHE STRING "SDL3 version fetched when no SDL3 is installed")
set(FE8_IMGUI_TAG "v1.92.9b" CACHE STRING "Dear ImGui version (options menu)")

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

string(TOUPPER "${FE8_REGION}" FE8_REGION_UPPER)

# ---------------------------------------------------------------------------
# Game code (the decomp), compiled natively.
# ---------------------------------------------------------------------------
file(GLOB FE8_GAME_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/src/*.c")
# src/msg_data.c is generated by the GBA build from texts/: the port uses the
# ROM's own text (gMsgTable is a ROM symbol).
list(FILTER FE8_GAME_SOURCES EXCLUDE REGEX "/src/msg_data\\.c$")
file(GLOB FE8_PORT_GAME_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/port/game/*.c")

# Absolute symbols: the ROM data the game reads in place, the RAM variables
# that stay at their GBA address, and the GBA memory regions.
set(FE8_SYMBOL_LISTS
    "${CMAKE_SOURCE_DIR}/port/data/rom_symbols_${FE8_REGION}.txt"
    "${CMAKE_SOURCE_DIR}/port/data/ram_symbols_${FE8_REGION}.txt"
    "${CMAKE_SOURCE_DIR}/port/data/hardware_symbols.txt")
if(FE8_REGION STREQUAL "eu")
    # Data only the European version has (per-language tables).
    file(GLOB FE8_EU_ONLY_LISTS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/port/data/eu_only/*.txt")
    list(APPEND FE8_SYMBOL_LISTS ${FE8_EU_ONLY_LISTS})
endif()
set(FE8_ABS_SYMBOLS "${CMAKE_BINARY_DIR}/gen/abs_symbols.S")
add_custom_command(
    OUTPUT "${FE8_ABS_SYMBOLS}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/gen"
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/tools/port/gen_abs_symbols.py"
            "${FE8_ABS_SYMBOLS}" ${FE8_SYMBOL_LISTS}
    DEPENDS ${FE8_SYMBOL_LISTS} "${CMAKE_SOURCE_DIR}/port/data/native_symbols.txt"
            "${CMAKE_SOURCE_DIR}/tools/port/gen_abs_symbols.py"
    VERBATIM)

# GBA address -> native function, for the function pointers in ROM data.
set(FE8_GBA_FUNCS "${CMAKE_BINARY_DIR}/gen/gba_funcs.c")
add_custom_command(
    OUTPUT "${FE8_GBA_FUNCS}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/gen"
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/tools/port/gen_gba_funcs.py"
            "${FE8_GBA_FUNCS}" "${CMAKE_SOURCE_DIR}/port/data/rom_funcs_${FE8_REGION}.txt"
    DEPENDS "${CMAKE_SOURCE_DIR}/port/data/rom_funcs_${FE8_REGION}.txt"
            "${CMAKE_SOURCE_DIR}/tools/port/gen_gba_funcs.py"
    VERBATIM)

# Game data the port reads from the ROM (see tools/port/dump_rom_symbols.py):
# sources that only hold it are left out, and in the ones that also hold code
# its definitions are only weak placeholders (PORT_ROM_EVENT_SCRIPTS).
list(FILTER FE8_GAME_SOURCES EXCLUDE REGEX
    "/src/(events_info|events_wm|events_trapdata|events_shoplist|events_udefs|data_battlequotes|data_characters|data_classes|data_items|data_event_trigger|data_terrains|data_supports|data_itembonus|data_itemuse|data_banimconf)\\.c$")
foreach(src events_script events_script_utils eventscr bmtrade sio_battlemap sio_event)
    set_source_files_properties("${CMAKE_SOURCE_DIR}/src/${src}.c"
        PROPERTIES COMPILE_DEFINITIONS PORT_ROM_EVENT_SCRIPTS)
endforeach()

add_library(fe8_game STATIC ${FE8_GAME_SOURCES} ${FE8_PORT_GAME_SOURCES} "${FE8_ABS_SYMBOLS}"
    "${FE8_GBA_FUNCS}")
target_include_directories(fe8_game PRIVATE
    "${CMAKE_SOURCE_DIR}/include" "${CMAKE_SOURCE_DIR}/port/generated" "${CMAKE_SOURCE_DIR}")
target_compile_definitions(fe8_game PRIVATE PLATFORM_PC NONMATCHING=1 REGION_${FE8_REGION_UPPER})
set(FE8_GAME_FLAGS
    -std=gnu11
    -fno-strict-aliasing      # GBA code type-puns freely
    -fwrapv                   # ...and relies on two's-complement overflow
    -funsigned-char           # char is unsigned on ARM
    -fcommon
    -fgnu89-inline            # agbcc semantics: a plain `inline` definition is external
    ${FE8_GAME_WARNINGS})
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
    # A few functions `return;` without a value (callers ignore the result).
    list(APPEND FE8_GAME_FLAGS -Wno-error=return-type -Werror=implicit-function-declaration
        -Werror=int-conversion -Werror=incompatible-pointer-types
        -Wno-error=incompatible-function-pointer-types)
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64")
    # The game's globals and ROM tables sit at their GBA addresses, which are
    # only 4-byte aligned, but on x86-64 the compiler assumes 16 for arrays of
    # 16+ bytes and uses aligned SSE loads on them (a crash in optimized
    # builds). The GBA code needs no floating point or vectors: no SSE.
    list(APPEND FE8_GAME_FLAGS -mno-sse)
endif()
target_compile_options(fe8_game PRIVATE
    $<$<COMPILE_LANGUAGE:C>:-include port/native.h ${FE8_GAME_FLAGS}>)

# ---------------------------------------------------------------------------
# Port layer.
# ---------------------------------------------------------------------------
add_library(fe8_port STATIC
    port/src/assets.c
    port/src/sha1.c
    port/src/gba_mem.c
    port/src/dma.c
    port/src/bios.c
    port/src/apu.c
    port/src/frame.c
    port/src/main.c
    port/src/savestate.c
    port/src/second_screen.c
    port/src/widescreen.c
    port/src/rumble.c
    port/src/pacing.c
    port/ppu/src/virtuappu.c
    port/ppu/src/mode1.c)
target_include_directories(fe8_port PUBLIC port/include PRIVATE port/ppu/include)
file(READ "${CMAKE_SOURCE_DIR}/fe8_${FE8_REGION}.sha1" FE8_SHA1_LINE)
string(REGEX MATCH "^[0-9a-fA-F]+" FE8_ROM_SHA1 "${FE8_SHA1_LINE}")
string(TOLOWER "${FE8_ROM_SHA1}" FE8_ROM_SHA1)
target_compile_definitions(fe8_port PRIVATE
    FE8_ROM_SHA1="${FE8_ROM_SHA1}" FE8_ROM_REGION_NAME="${FE8_REGION_UPPER}")
target_compile_options(fe8_port PRIVATE $<$<COMPILE_LANGUAGE:C>:-std=gnu11 -Wall -Wextra -Wno-unused-parameter>)
if(NOT ANDROID)
    find_package(OpenMP COMPONENTS C)
    if(OpenMP_C_FOUND)
        target_link_libraries(fe8_port PRIVATE OpenMP::OpenMP_C)
    endif()
endif()
target_link_libraries(fe8_port PUBLIC m)
if(ANDROID)
    target_link_libraries(fe8_port PUBLIC log)
endif()

# The game image is placed below 4 GiB: a non-PIE executable at
# FE8_IMAGE_BASE on desktop (Android loads its library there at run time).
function(fe8_low_image target)
    if(NOT ANDROID)
        set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE OFF)
        target_link_options(${target} PRIVATE -no-pie -fuse-ld=lld "-Wl,--image-base=${FE8_IMAGE_BASE}")
    endif()
endfunction()
if(NOT ANDROID)
    foreach(lib fe8_game fe8_port)
        set_target_properties(${lib} PROPERTIES POSITION_INDEPENDENT_CODE OFF)
        target_compile_options(${lib} PRIVATE -fno-pic -fno-pie)
    endforeach()
endif()

# ---------------------------------------------------------------------------
# Executables.
# ---------------------------------------------------------------------------
if(FE8_HOST STREQUAL "headless")
    add_executable(fe8_headless port/src/host_headless.c)
    target_compile_options(fe8_headless PRIVATE -std=gnu11 -Wall -Wextra -fno-pic -fno-pie)
    target_link_libraries(fe8_headless PRIVATE fe8_port fe8_game fe8_port)
    fe8_low_image(fe8_headless)
    # Frame pacing / sound rate control test (no ROM needed).
    add_executable(fe8_pacing_test tools/port/pacing_test.c port/src/pacing.c)
    target_include_directories(fe8_pacing_test PRIVATE port/include)
    target_compile_options(fe8_pacing_test PRIVATE -std=gnu11 -Wall -Wextra)
    target_link_libraries(fe8_pacing_test PRIVATE m)
elseif(FE8_HOST STREQUAL "sdl")
    if(NOT ANDROID)
        find_package(SDL3 CONFIG QUIET)
    endif()
    if(NOT TARGET SDL3::SDL3)
        # No system SDL3 (always the case on Android): build it from source.
        include(FetchContent)
        set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
        set(SDL_TESTS OFF CACHE BOOL "" FORCE)
        set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
        FetchContent_Declare(SDL3
            GIT_REPOSITORY https://github.com/libsdl-org/SDL.git
            GIT_TAG ${FE8_SDL3_TAG}
            GIT_SHALLOW TRUE)
        FetchContent_MakeAvailable(SDL3)
    endif()
    # Options menu: Dear ImGui with its SDL3 + SDL_Renderer backends.
    enable_language(CXX)
    include(FetchContent)
    FetchContent_Declare(imgui
        GIT_REPOSITORY https://github.com/ocornut/imgui.git
        GIT_TAG ${FE8_IMGUI_TAG}
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(imgui)
    add_library(fe8_imgui STATIC
        ${imgui_SOURCE_DIR}/imgui.cpp
        ${imgui_SOURCE_DIR}/imgui_draw.cpp
        ${imgui_SOURCE_DIR}/imgui_tables.cpp
        ${imgui_SOURCE_DIR}/imgui_widgets.cpp
        ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl3.cpp
        ${imgui_SOURCE_DIR}/backends/imgui_impl_sdlrenderer3.cpp)
    target_include_directories(fe8_imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
    target_compile_definitions(fe8_imgui PUBLIC IMGUI_DISABLE_OBSOLETE_FUNCTIONS IMGUI_DISABLE_DEMO_WINDOWS
                                               IMGUI_DISABLE_DEBUG_TOOLS)
    target_link_libraries(fe8_imgui PUBLIC SDL3::SDL3)

    set(FE8_SDL_SOURCES port/src/host_sdl.c port/src/config.c port/src/menu.cpp)
    if(ANDROID)
        # SDLActivity loads libmain.so and calls its SDL_main, which loads the
        # game (libgame.so) at a reserved address below 4 GiB and runs it.
        add_library(game SHARED ${FE8_SDL_SOURCES})
        set(FE8_EXE game)
        add_library(main SHARED port/src/android_launcher.c)
        target_link_libraries(main PRIVATE dl log)
        target_compile_definitions(main PRIVATE FE8_IMAGE_BASE=${FE8_IMAGE_BASE})
        add_dependencies(main game)
    else()
        add_executable(fe8 ${FE8_SDL_SOURCES})
        set(FE8_EXE fe8)
        fe8_low_image(fe8)
    endif()
    target_compile_options(${FE8_EXE} PRIVATE $<$<COMPILE_LANGUAGE:C>:-std=gnu11> -Wall -Wextra)
    target_compile_features(${FE8_EXE} PRIVATE cxx_std_17)
    target_link_libraries(${FE8_EXE} PRIVATE fe8_port fe8_game fe8_port fe8_imgui SDL3::SDL3)
else()
    message(FATAL_ERROR "FE8_HOST must be sdl or headless")
endif()
